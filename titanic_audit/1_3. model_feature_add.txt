# Train에서 10명 이상인 자주 등장하는 Title만 그대로 유지합니다.
# 희귀한 Title은 모델이 너무 작은 집단을 외우지 않도록 Rare로 묶습니다.
train_title_counts = train_title_preview.value_counts()
common_titles = train_title_counts[train_title_counts >= 10].index.tolist()
print("그대로 유지할 Title:", common_titles)


def add_features(data):
    """Train과 Test에 요청한 다섯 가지 Feature를 같은 방법으로 추가합니다."""

    # 원본 데이터를 직접 바꾸지 않도록 복사합니다.
    data = data.copy()

    # sibsp는 함께 탑승한 형제/배우자 수입니다.
    # parch는 함께 탑승한 부모/자녀 수입니다.
    # +1은 승객 본인을 의미합니다.
    data["FamilySize"] = (
        data["sibsp"]
        + data["parch"]
        + 1
    )

    # 가족 수가 1명이면 혼자 탑승한 승객이므로 1로 표시합니다.
    data["IsAlone"] = (
        data["FamilySize"] == 1
    ).astype(int)

    # 이름의 쉼표 뒤부터 마침표 앞까지를 원래 Title로 추출합니다.
    original_title = (
        data["name"]
        .str.extract(r",\s*([^.]*)\.")[0]
        .str.strip()
    )

    # 직업이나 사회적 역할을 나타내는 Title만 간단히 구분합니다.
    # Mr, Mrs, Miss, Master 등 일반 호칭은 Other로 둡니다.
    data["Occupation"] = "Other"
    data.loc[original_title == "Dr", "Occupation"] = "Doctor"
    data.loc[original_title == "Rev", "Occupation"] = "Clergy"
    data.loc[
        original_title.isin(["Col", "Major", "Capt"]),
        "Occupation",
    ] = "Military"

    # 실제 Train/Test에서 확인된 귀족 또는 상류계급 관련 Title입니다.
    noble_titles = [
        "Sir",
        "Lady",
        "the Countess",
        "Don",
        "Dona",
        "Jonkheer",
    ]
    data["IsNoble"] = original_title.isin(noble_titles).astype(int)

    # 자주 등장하는 Title은 유지하고 나머지 희귀 Title은 Rare로 묶습니다.
    # Occupation과 IsNoble은 묶기 전 원래 Title에서 이미 계산했습니다.
    data["Title"] = original_title
    data.loc[~data["Title"].isin(common_titles), "Title"] = "Rare"

    return data

def cross_validation(features, targets, cat_cols, use_smote=False):
    """2번 모델과 같은 방식으로 Stratified 5-Fold 교차검증을 수행합니다."""

    # 전체 데이터를 target 비율이 유지되도록 5개의 Fold로 나눕니다.
    stratified_kfold = StratifiedKFold(
        n_splits=5,
        shuffle=True,
        random_state=0,
    )

    # 교차검증 결과를 저장할 리스트입니다.
    auc_scores = []
    accuracy_scores = []

    # 각 행이 학습에 포함되지 않았던 Fold에서 받은 OOF 예측을 저장합니다.
    oof_pred = np.zeros(len(features), dtype=int)
    oof_proba = np.zeros(len(features), dtype=float)

    if use_smote:
        experiment_name = "3. model_feature_add + SMOTE"
    else:
        experiment_name = "3. model_feature_add"

    print(f"[{experiment_name} Cross Validation]")

    # 전체 데이터를 Fold별 Train과 Validation으로 분리합니다.
    for fold, (train_idx, valid_idx) in enumerate(
        stratified_kfold.split(features, targets),
        start=1,
    ):
        # 현재 Fold에서 사용할 학습 데이터입니다.
        x_train_fold = features.iloc[train_idx].copy()
        y_train_fold = targets.iloc[train_idx].copy()

        # 원본 Validation 데이터에는 SMOTE를 적용하지 않습니다.
        x_valid_fold = features.iloc[valid_idx].copy()
        y_valid_fold = targets.iloc[valid_idx].copy()

        if use_smote:
            # Fold Train에서만 범주형 인코더를 학습합니다.
            # Title과 Occupation도 cat_cols에 있어 함께 변환됩니다.
            category_encoder = OrdinalEncoder(
                handle_unknown="use_encoded_value",
                unknown_value=-1,
            )
            x_train_fold[cat_cols] = category_encoder.fit_transform(
                x_train_fold[cat_cols]
            )
            x_valid_fold[cat_cols] = category_encoder.transform(
                x_valid_fold[cat_cols]
            )

            # SMOTE 적용 전 Fold Train의 클래스 개수입니다.
            print(f"\nFold {fold} - SMOTE 적용 전")
            print(y_train_fold.value_counts().sort_index())

            # 현재 Fold의 Train에만 SMOTE를 적용합니다.
            smote = SMOTE(random_state=0)
            x_train_fold, y_train_fold = smote.fit_resample(
                x_train_fold,
                y_train_fold,
            )

            # 2번 모델과 동일하게 범주형 열을 가장 가까운 범주 코드로 되돌립니다.
            x_train_fold[cat_cols] = x_train_fold[cat_cols].round().astype("int64")
            x_valid_fold[cat_cols] = x_valid_fold[cat_cols].astype("int64")

            # SMOTE 적용 후 Fold Train의 클래스 개수입니다.
            print(f"Fold {fold} - SMOTE 적용 후")
            print(y_train_fold.value_counts().sort_index())

        # Fold마다 기존 Modeling 객체를 새로 생성합니다.
        fold_modeling = Modeling(
            x_tr=x_train_fold,
            y_tr=y_train_fold,
            cat_cols=cat_cols,
        )

        # Fold Train으로 학습하고 원본 Fold Validation으로 평가합니다.
        fold_modeling.fit_evaluation(
            x_te=x_valid_fold,
            y_te=y_valid_fold,
        )

        best_model_info = fold_modeling.get_best_model()
        valid_pred = fold_modeling.predict_by_best_model(x_valid_fold)
        
        # survived=1의 확률값은 ROC-AUC 계산에 사용합니다.
        valid_proba = fold_modeling.predict_proba_by_best_model(x_valid_fold)

        # ROC-AUC에는 확률값을 사용합니다.
        fold_auc = get_auc_score(y=y_valid_fold, pred=valid_proba)

        # Accuracy에는 0/1 예측값을 사용합니다.
        fold_accuracy = accuracy_score(y_valid_fold, valid_pred)

        auc_scores.append(fold_auc)
        accuracy_scores.append(fold_accuracy)

        # 현재 Fold의 Validation 위치에 OOF 결과를 저장합니다.
        oof_pred[valid_idx] = valid_pred
        oof_proba[valid_idx] = valid_proba

        print(f"\nFold {fold}:")
        print(f"Train 데이터 크기: {x_train_fold.shape}")
        print(f"Validation 데이터 크기: {x_valid_fold.shape}")
        print(f"Best Model: {best_model_info['model_name']}")
        print(f"AUC: {fold_auc:.6f}")
        print(f"Accuracy: {fold_accuracy:.6f}")
        print("-" * 50)

    # 5개 Fold의 점수와 평균, 표준편차를 출력합니다.
    print(f"\n[{experiment_name} Cross Validation 결과]")
    for fold, (auc_score, accuracy) in enumerate(
        zip(auc_scores, accuracy_scores),
        start=1,
    ):
        print(f"Fold {fold}: AUC={auc_score:.6f}, Accuracy={accuracy:.6f}")

    print(f"AUC Mean: {np.mean(auc_scores):.6f}")
    print(f"AUC Std: {np.std(auc_scores):.6f}")
    print(f"Accuracy Mean: {np.mean(accuracy_scores):.6f}")
    print(f"Accuracy Std: {np.std(accuracy_scores):.6f}")

    return {
        "auc_scores": auc_scores,
        "accuracy_scores": accuracy_scores,
        "auc_mean": np.mean(auc_scores),
        "auc_std": np.std(auc_scores),
        "accuracy_mean": np.mean(accuracy_scores),
        "accuracy_std": np.std(accuracy_scores),
        "oof_pred": oof_pred,
        "oof_proba": oof_proba,
    }
