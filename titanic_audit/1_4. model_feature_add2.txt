def add_title_features(data):
    """3번 실험과 같은 방법으로 이름 관련 Feature를 만듭니다."""

    data = data.copy()

    # 이름의 쉼표 뒤부터 마침표 앞까지를 원래 Title로 추출합니다.
    original_title = (
        data["name"]
        .str.extract(r",\s*([^.]*)\.")[0]
        .str.strip()
    )

    # 직업이나 사회적 역할을 나타내는 Title만 간단히 구분합니다.
    data["Occupation"] = "Other"
    data.loc[original_title == "Dr", "Occupation"] = "Doctor"
    data.loc[original_title == "Rev", "Occupation"] = "Clergy"
    data.loc[
        original_title.isin(["Col", "Major", "Capt"]),
        "Occupation",
    ] = "Military"

    # 실제 Train/Test에서 확인된 귀족 또는 상류계급 관련 Title입니다.
    noble_titles = [
        "Sir", "Lady", "the Countess",
        "Don", "Dona", "Jonkheer",
    ]
    data["IsNoble"] = original_title.isin(noble_titles).astype(int)

    # 자주 등장하는 Title은 유지하고 나머지는 Rare로 묶습니다.
    data["Title"] = original_title
    data.loc[~data["Title"].isin(common_titles), "Title"] = "Rare"

    return data

# Age 대체 통계량은 결측치가 남아 있는 원본 Train에서만 계산합니다.
age_median_by_title_pclass = train.groupby(
    ["Title", "pclass"]
)["age"].median()
age_median_by_title = train.groupby("Title")["age"].median()
age_median_all = train["age"].median()


def fill_age_from_train(data):
    """Train에서 계산한 중앙값만 사용하여 Age 결측치를 채웁니다."""

    data = data.copy()
    missing_age_indexes = data.index[data["age"].isna()]

    for index in missing_age_indexes:
        title = data.at[index, "Title"]
        pclass = data.at[index, "pclass"]

        # 첫 번째 선택: 같은 Title과 pclass를 가진 Train 승객의 중앙값
        fill_value = age_median_by_title_pclass.get((title, pclass), np.nan)

        # 두 번째 선택: 같은 Title을 가진 Train 승객의 중앙값
        if pd.isna(fill_value):
            fill_value = age_median_by_title.get(title, np.nan)

        # 마지막 선택: Train 전체 Age 중앙값
        if pd.isna(fill_value):
            fill_value = age_median_all

        data.at[index, "age"] = fill_value

    return data

# 기존 FamilySize와 IsAlone Feature를 그대로 만듭니다.
for frame in [train, test]:
    # sibsp + parch + 승객 본인 1명으로 전체 가족 수를 계산합니다.
    frame["FamilySize"] = (
        frame["sibsp"]
        + frame["parch"]
        + 1
    )

    # 가족 수가 1명이면 혼자 탑승한 승객입니다.
    frame["IsAlone"] = (frame["FamilySize"] == 1).astype(int)

    # Ticket 그룹 인원수를 고려한 1인당 요금을 계산합니다.
    # TicketGroupSize가 0보다 큰지 위에서 확인했으므로 0으로 나누지 않습니다.
    frame["FarePerPerson"] = (
        frame["fare"] / frame["TicketGroupSize"]
    )

# FarePerPerson에 NaN, inf, -inf가 없는지 확인합니다.

def cross_validation(features, targets, cat_cols):
    """Fold Train에만 SMOTE를 적용하고 CatBoost로 평가합니다."""

    stratified_kfold = StratifiedKFold(
        n_splits=5,
        shuffle=True,
        random_state=0,
    )

    auc_scores = []
    accuracy_scores = []

    # 각 행이 학습에 포함되지 않았던 Fold의 OOF 결과를 저장합니다.
    oof_pred = np.zeros(len(features), dtype=int)
    oof_proba = np.zeros(len(features), dtype=float)

    print("[Feature Add2 + SMOTE + CatBoost Cross Validation]")

    for fold, (train_idx, valid_idx) in enumerate(
        stratified_kfold.split(features, targets),
        start=1,
    ):
        x_train_fold = features.iloc[train_idx].copy()
        y_train_fold = targets.iloc[train_idx].copy()

        # Validation 데이터에는 SMOTE를 적용하지 않습니다.
        x_valid_fold = features.iloc[valid_idx].copy()
        y_valid_fold = targets.iloc[valid_idx].copy()
        original_valid_size = len(x_valid_fold)

        # 범주형 인코더는 현재 Fold Train에서만 학습합니다.
        category_encoder = OrdinalEncoder(
            handle_unknown="use_encoded_value",
            unknown_value=-1,
        )
        x_train_fold[cat_cols] = category_encoder.fit_transform(
            x_train_fold[cat_cols]
        )
        x_valid_fold[cat_cols] = category_encoder.transform(
            x_valid_fold[cat_cols]
        )

        print(f"\nFold {fold}")
        print("SMOTE 적용 전 클래스 개수")
        print(y_train_fold.value_counts().sort_index())

        # 현재 Fold Train에만 SMOTE를 적용합니다.
        smote = SMOTE(random_state=0)
        x_train_fold, y_train_fold = smote.fit_resample(
            x_train_fold,
            y_train_fold,
        )

        # SMOTE가 만든 범주형 열의 소수값을 가장 가까운 정수 코드로 복구합니다.
        x_train_fold[cat_cols] = x_train_fold[cat_cols].round().astype("int64")
        x_valid_fold[cat_cols] = x_valid_fold[cat_cols].astype("int64")

        print("SMOTE 적용 후 클래스 개수")
        print(y_train_fold.value_counts().sort_index())

        # common의 CatBoost 기본 설정과 같이 verbose=0, cat_features를 사용합니다.
        fold_model = CatBoostClassifier(
            verbose=0,
            cat_features=cat_cols,
        )
        fold_model.fit(x_train_fold, y_train_fold)

        # ROC-AUC는 survived=1 확률, Accuracy는 0/1 예측값을 사용합니다.
        valid_proba = fold_model.predict_proba(x_valid_fold)[:, 1]
        valid_pred = fold_model.predict(x_valid_fold).astype(int).reshape(-1)

        fold_auc = get_auc_score(y=y_valid_fold, pred=valid_proba)
        fold_accuracy = accuracy_score(y_valid_fold, valid_pred)

        auc_scores.append(fold_auc)
        accuracy_scores.append(fold_accuracy)
        oof_pred[valid_idx] = valid_pred
        oof_proba[valid_idx] = valid_proba

        print(f"Train 크기: {x_train_fold.shape}")
        print(f"Validation 크기: {x_valid_fold.shape}")
        print(f"CatBoost AUC: {fold_auc:.6f}")
        print(f"CatBoost Accuracy: {fold_accuracy:.6f}")
        assert len(x_valid_fold) == original_valid_size
        print("-" * 50)

    print("\n[Feature Add2 + SMOTE + CatBoost CV 결과]")
    for fold, (auc_score, accuracy) in enumerate(
        zip(auc_scores, accuracy_scores),
        start=1,
    ):
        print(f"Fold {fold}: AUC={auc_score:.6f}, Accuracy={accuracy:.6f}")

    print(f"AUC Mean: {np.mean(auc_scores):.6f}")
    print(f"AUC Std: {np.std(auc_scores):.6f}")
    print(f"Accuracy Mean: {np.mean(accuracy_scores):.6f}")
    print(f"Accuracy Std: {np.std(accuracy_scores):.6f}")

    return {
        "auc_scores": auc_scores,
        "accuracy_scores": accuracy_scores,
        "auc_mean": np.mean(auc_scores),
        "auc_std": np.std(auc_scores),
        "accuracy_mean": np.mean(accuracy_scores),
        "accuracy_std": np.std(accuracy_scores),
        "oof_pred": oof_pred,
        "oof_proba": oof_proba,
    }

# 10. 전체 Train + SMOTE + CatBoost 최종 학습

### CELL 37 [code] execution=19
# 범주형 인코더는 전체 Train에서만 학습하고 Test에는 transform만 적용합니다.
final_encoder = OrdinalEncoder(
    handle_unknown="use_encoded_value",
    unknown_value=-1,
)

x_train_final = features.copy()
x_test_final = test.copy()
x_train_final[cat_cols] = final_encoder.fit_transform(x_train_final[cat_cols])
x_test_final[cat_cols] = final_encoder.transform(x_test_final[cat_cols])

print("전체 Train - SMOTE 적용 전")
print(target.value_counts().sort_index())

# 최종 학습용 전체 Train에만 SMOTE를 적용합니다. Test에는 적용하지 않습니다.
final_smote = SMOTE(random_state=0)
x_train_final, y_train_final = final_smote.fit_resample(
    x_train_final,
    target,
)
x_train_final[cat_cols] = x_train_final[cat_cols].round().astype("int64")
x_test_final[cat_cols] = x_test_final[cat_cols].astype("int64")

print("\n전체 Train - SMOTE 적용 후")
print(y_train_final.value_counts().sort_index())
print("Test 크기 유지:", x_test_final.shape)

# 여러 모델을 비교하지 않고 CatBoost 하나만 최종 학습합니다.
final_model = CatBoostClassifier(
    verbose=0,
    cat_features=cat_cols,
)
final_model.fit(x_train_final, y_train_final)
print("최종 CatBoost 학습 완료")
